from transformers import pipeline
import torch

generator = pipeline(
    "text-generation",
    model="bigcode/starcoder2-3b",
    device_map="auto",
    torch_dtype=torch.float16
)

prompt = "Python program to add two numbers"

result = generator(
    prompt,
    max_new_tokens=50,
    do_sample=True
)

print(result[0]['generated_text'])
