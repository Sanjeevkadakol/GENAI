from openai import OpenAI

client = OpenAI(
    base_url="",
    api_key=""
)

response = client.chat.completions.create(
    model="openai/gpt-3.5-turbo",
    messages=[
        {
            "role": "user",
            "content": "Evaluate this resume: Python developer with AI skills"
        }
    ]
)

print(response.choices[0].message.content)
